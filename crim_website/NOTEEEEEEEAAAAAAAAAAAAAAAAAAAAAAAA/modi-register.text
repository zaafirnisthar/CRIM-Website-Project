<?php
// register.php (form, server validation + insert)
// Make sure connect.php is in the same folder.
require_once __DIR__ . '/../config/db.php';

// Prepare state
$errors = [];                 // associative array for field errors
$values = ['fullname'=>'','email'=>'','username'=>'']; // keep entered values
$success = false;

if ($_SERVER['REQUEST_METHOD'] === 'POST') {
    // collect inputs safely
    $fullname = trim($_POST['fullname'] ?? '');
    $email    = trim($_POST['email'] ?? '');
    $username = trim($_POST['username'] ?? '');
    $password = $_POST['password'] ?? '';
    $confirm  = $_POST['confirm'] ?? '';

    // keep values to show back in the form (don't repopulate password fields)
    $values['fullname'] = $fullname;
    $values['email']    = $email;
    $values['username'] = $username;

    // server-side validations
    if ($password !== $confirm) {
        $errors['password'] = "Passwords do not match!";
    }

    if ($email === '') {
        $errors['email'] = "Email is required.";
    } elseif (!filter_var($email, FILTER_VALIDATE_EMAIL)) {
        $errors['email'] = "Invalid email format.";
    }

    // check duplicate email only if no earlier email error
    if (!isset($errors['email'])) {
        $stmt = $conn->prepare("SELECT id FROM users WHERE email = ? LIMIT 1");
        if ($stmt) {
            $stmt->bind_param("s", $email);
            $stmt->execute();
            $stmt->store_result();
            if ($stmt->num_rows > 0) {
                $errors['email'] = "Email already registered!";
            }
            $stmt->close();
        } else {
            $errors['general'] = "Database error (check SELECT).";
        }
    }

    // If no errors, insert user
    if (empty($errors)) {
        $hashed = password_hash($password, PASSWORD_DEFAULT);
        $ins = $conn->prepare("INSERT INTO users (fullname, email, username, password) VALUES (?, ?, ?, ?)");
        if ($ins) {
            $ins->bind_param("ssss", $fullname, $email, $username, $hashed);
            if ($ins->execute()) {
                $success =  "Registration successful. You can now <a href='login.php'> login </a>.";
                // clear shown values after success
                $values = ['fullname'=>'','email'=>'','username'=>''];
            } else {
                $errors['general'] = "Database error (insert): " . $ins->error;
            }
            $ins->close();
        } else {
            $errors['general'] = "Database error (prepare insert).";
        }
    }
}

  document.getElementById('regForm').addEventListener('submit', function(e) {
    var pass = document.getElementById('password').value;
